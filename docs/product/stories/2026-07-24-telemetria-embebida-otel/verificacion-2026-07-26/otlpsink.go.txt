// otlpsink — receptor OTLP mínimo de verificación (loopback).
// Vuelca cada POST a disco tal cual llega, sin decodificar. Sirve para contestar
// empíricamente: qué señales manda Claude Code, en qué endpoint, con qué atributos.
package main

import (
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"sync/atomic"
)

var n atomic.Int64

func main() {
	dir := os.Args[1]
	if err := os.MkdirAll(dir, 0o755); err != nil {
		panic(err)
	}
	h := func(kind string) http.HandlerFunc {
		return func(w http.ResponseWriter, r *http.Request) {
			b, _ := io.ReadAll(r.Body)
			i := n.Add(1)
			name := fmt.Sprintf("%03d-%s-%s.bin", i, kind, sanitize(r.Header.Get("Content-Type")))
			_ = os.WriteFile(filepath.Join(dir, name), b, 0o644)
			fmt.Printf("%s %s ct=%q bytes=%d\n", r.Method, r.URL.Path, r.Header.Get("Content-Type"), len(b))
			w.Header().Set("Content-Type", r.Header.Get("Content-Type"))
			w.WriteHeader(http.StatusOK)
			_, _ = w.Write([]byte("{}"))
		}
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/v1/metrics", h("metrics"))
	mux.HandleFunc("/v1/logs", h("logs"))
	mux.HandleFunc("/v1/traces", h("traces"))
	mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		fmt.Printf("OTRO %s %s\n", r.Method, r.URL.Path)
		w.WriteHeader(http.StatusOK)
	})
	fmt.Println("otlpsink escuchando 127.0.0.1:4318 →", dir)
	_ = http.ListenAndServe("127.0.0.1:4318", mux)
}

func sanitize(s string) string {
	out := []rune{}
	for _, r := range s {
		if r == '/' || r == ';' || r == ' ' {
			r = '_'
		}
		out = append(out, r)
	}
	return string(out)
}
